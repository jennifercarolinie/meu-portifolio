// TIMER
let tempo = 90*60;
setInterval(()=>{ if(tempo > 0) tempo--; }, 1000);

// BANCO DE DADOS LOCAL
let bancoDeNotas = JSON.parse(localStorage.getItem('notasVersionamento')) || [];

document.getElementById('formAvaliacao').addEventListener('submit', e=>{
    e.preventDefault();
    const dados = new FormData(e.target);
    let acertos = 0; let feedback = ''; let respostas = [];

    questoes.forEach((questao, i)=>{
        let resposta = parseInt(dados.get('q'+i));
        respostas.push(questao.op[resposta] || 'Em branco');
        if(resposta === questao.r){ acertos++; feedback += `<div class="feedback certo">Q${i+1}: ✅ Acertou [${questao.tag}]</div>`; }
        else { feedback += `<div class="feedback errado">Q${i+1}: ❌ Errou [${questao.tag}]. Certa: ${questao.op[questao.r]}</div>`; }
    });

    let nota = ((acertos/32)*10).toFixed(1);
    let tempoGasto = 90*60 - tempo;

    const aluno = {
        data: new Date().toLocaleString('pt-BR'),
        nome: dados.get('nome'),
        turma: dados.get('turma'),
        nota: nota,
        acertos: acertos,
        tempo: `${Math.floor(tempoGasto/60)}m ${tempoGasto%60}s`,
      ...Object.fromEntries(respostas.map((r,i)=>[`Q${i+1}`, r])),
        D33: dados.get('q33'), D34: dados.get('q34')
    };
    bancoDeNotas.push(aluno);
    localStorage.setItem('notasVersionamento', JSON.stringify(bancoDeNotas));

    document.getElementById('formAvaliacao').style.display = 'none';
    document.getElementById('resultado').style.display = 'block';
    document.getElementById('placar').innerHTML = `Aluno: ${aluno.nome}<br>Nota: <span class="acerto">${nota}</span> / 10.0<br>Acertos: ${acertos}/32`;
    document.getElementById('feedback-geral').innerHTML = feedback;
});

// EXPORTAR CSV
function exportarCSV(){
    if(bancoDeNotas.length === 0) return alert('Nenhuma avaliação salva ainda!');

    let headers = ['Data','Nome','Turma','Nota','Acertos','Tempo'];
    questoes.forEach((_,i)=> headers.push(`Q${i+1}`));
    headers.push('D33','D34');

    let csv = [headers.join(';')];
    bancoDeNotas.forEach(aluno=>{
        let linha = headers.map(h => `"${aluno[h] || ''}"`);
        csv.push(linha.join(';'));
    });

    const blob = new Blob([csv.join('\n')], { type: 'text/csv;charset=utf-8;' });
    const url = URL.createObjectURL(blob);
    const link = document.createElement('a');
    link.href = url;
    link.download = `Notas_Versionamento_${new Date().toLocaleDateString('pt-BR')}.csv`;
    link.click();
}
